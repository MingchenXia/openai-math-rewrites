\title{Maximal multipoint Seshadri constants in higher dimensions}
\author{}
\date{}

\begin{document}

\begin{abstract}
Let $L$ be an ample line bundle on a smooth integral complex projective variety $X$ of dimension $n\geq3$. For all sufficiently large $r$, we prove that the multipoint Seshadri constant of $L$ at $r$ very general points is $(L^n/r)^{1/n}$. This gives the qualitative Nagata--Biran--Szemberg bound in higher dimensions.
\end{abstract}
