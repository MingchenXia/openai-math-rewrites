\title{Nagata's conjecture for plane curves}
\author{}
\date{}

\begin{document}

\begin{abstract}
We prove that for $r\geq10$ very general points of the complex projective plane, the sum of the multiplicities of any nonzero effective curve of degree $d$ is strictly less than $d\sqrt r$. The proof reduces unequal multiplicities to equal ones by symmetrization and treats the nonsquare case by degeneration on the normal bundle of a smooth cubic.
\end{abstract}
