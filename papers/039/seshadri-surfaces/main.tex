\title{Maximal Seshadri constants on arbitrary polarized surfaces}
\author{}
\date{}

\begin{document}

\begin{abstract}
Let $L$ be an ample line bundle on a smooth integral complex projective surface $S$. We prove that the multipoint Seshadri constant of $L$ at $r$ very general points is $\sqrt{L^2/r}$ for all sufficiently large $r$. The proof transfers injective jet tests from finite subgroups of a two-dimensional torus.
\end{abstract}
