\title{The symmetric Mahler conjecture and its equality cases}
\author{}
\date{}

\begin{document}

\begin{abstract}
We prove that every origin-symmetric convex body in $\mathbb{R}^n$ has volume product at least $4^n/n!$, with equality exactly for invertible linear images of Hanner polytopes. We also obtain the sharp even functional Mahler inequality.
\end{abstract}
