\title{The Mahler conjecture for general convex bodies}
\author{}
\date{}

\begin{document}

\begin{abstract}
We prove that the volume product of a convex body in $\mathbb{R}^n$ about its Santal\'o point is at least $(n+1)^{n+1}/(n!)^2$. Equality holds precisely for simplices. We also prove the corresponding sharp functional Mahler inequality.
\end{abstract}
