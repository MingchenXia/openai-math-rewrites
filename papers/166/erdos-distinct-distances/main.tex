\title{The higher-dimensional Erd\H{o}s distinct-distances conjecture}
\author{}
\date{}

\begin{document}

\begin{abstract}
For every fixed $d\geq3$, we prove that $n\geq2$ distinct points in $\mathbb{R}^d$ determine at least $c_d n^{2/d}$ distances, with $c_d>0$ depending only on $d$. Integer grids show that the exponent $2/d$ is sharp.
\end{abstract}
